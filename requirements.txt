# Cyclone Track Forecasting — dependencies
#
# Upper bounds are deliberate. Model artifacts are pickled estimators, and
# unpickling one under a different scikit-learn version is a known source of
# quiet misbehaviour rather than a clean error. save_model.py records the
# versions that produced each artifact and warns on mismatch, but pinning the
# major version avoids the situation entirely.

# ---- Core data stack ----
numpy>=1.24,<3.0
pandas>=2.0,<3.0
scipy>=1.10

# ---- Storage ----
# Parquet is the primary interchange format between pipeline stages: it
# preserves dtypes (so ISO_TIME needs no re-parsing) and loads far faster than
# CSV. Required, not optional.
pyarrow>=14.0

# ---- Models ----
scikit-learn>=1.3,<2.0
xgboost>=3.0,<4.0
joblib>=1.3

# ---- Tuning ----
optuna>=3.4

# ---- Figures ----

# ---- Reporting ----

# report silently fails to generate.

uvicorn[standard]>=0.34
pydantic>=2.0

# ---- Dashboard ----
streamlit>=1.30

                       # so the track map works in an offline/sandboxed container
